from typing import Literal

from pydantic import ConfigDict, Field, field_validator, model_validator

from app.schemas.base import ContractModel
from app.schemas.enums import FaultType, ServiceType

SLUG_PATTERN = r"^[a-z0-9][a-z0-9_-]*$"
HEALTH_PREFIXES = ("/", "tcp://", "http://", "https://")


class AdapterService(ContractModel):
    name: str = Field(pattern=SLUG_PATTERN)
    container: str = Field(min_length=1)  # docker compose service name
    type: ServiceType
    critical: bool = False
    health: str | None = None
    faults: list[FaultType] = Field(default_factory=list)

    @field_validator("health")
    @classmethod
    def health_must_be_probeable(cls, v: str | None) -> str | None:
        if v is not None and not v.startswith(HEALTH_PREFIXES):
            raise ValueError(
                "health must be an HTTP path starting with '/', an http(s):// URL, or a 'tcp://host:port' address"
            )
        return v


class DeclaredDependency(ContractModel):
    # "from" is a Python keyword, so the field is from_ and the file key is "from".
    model_config = ConfigDict(populate_by_name=True)

    from_: str = Field(alias="from")
    to: str


class AdapterFile(ContractModel):
    app: str = Field(pattern=SLUG_PATTERN)
    version: Literal[1] = 1
    description: str | None = None
    compose_project: str | None = Field(default=None, pattern=SLUG_PATTERN)
    services: list[AdapterService] = Field(min_length=1)
    declared_dependencies: list[DeclaredDependency] = Field(default_factory=list)

    @model_validator(mode="after")
    def check_references(self) -> "AdapterFile":
        names = [s.name for s in self.services]
        duplicates = {n for n in names if names.count(n) > 1}
        if duplicates:
            raise ValueError(f"duplicate service names: {sorted(duplicates)}")
        known = set(names)
        for dep in self.declared_dependencies:
            for end in (dep.from_, dep.to):
                if end not in known:
                    raise ValueError(f"declared dependency references unknown service '{end}'")
        return self

from uuid import UUID, uuid4

from pydantic import Field, StrictBool, StrictFloat, StrictInt, StrictStr

from app.schemas.base import ContractModel
from app.schemas.enums import ActionType, AgentRole

# Contracts: parameters are Record<string, string | number | boolean>.
ParamValue = StrictBool | StrictInt | StrictFloat | StrictStr


class ActionProposal(ContractModel):
    proposal_id: UUID
    incident_id: UUID
    agent: AgentRole
    action: ActionType
    target_service: str
    parameters: dict[str, ParamValue] = Field(default_factory=dict)
    rationale: str  # display only; the gateway never evaluates it
    evidence_refs: list[UUID] = Field(default_factory=list)  # incident_event ids


class Hypothesis(ContractModel):
    service: str
    cause: str
    confidence: float = Field(ge=0, le=1)
    evidence_refs: list[UUID] = Field(default_factory=list)


class RootCauseReport(ContractModel):
    incident_id: UUID
    hypotheses: list[Hypothesis]


class CandidateAction(ContractModel):
    action: ActionType
    target_service: str
    blast_radius: list[str]


class ImpactReport(ContractModel):
    incident_id: UUID
    affected_services: list[str]
    candidate_actions: list[CandidateAction]
    summary: str


# --- Drafts (Contracts v0.2 section 3) ---------------------------------------
# The model returns a draft. The orchestrator adds the IDs, so an LLM never invents them.


class ActionProposalDraft(ContractModel):
    action: ActionType
    target_service: str
    parameters: dict[str, ParamValue] = Field(default_factory=dict)
    rationale: str
    evidence_refs: list[UUID] = Field(default_factory=list)

    def finalize(self, incident_id: UUID, agent: AgentRole) -> ActionProposal:
        return ActionProposal(
            proposal_id=uuid4(),
            incident_id=incident_id,
            agent=agent,
            **self.model_dump(),
        )


class RootCauseReportDraft(ContractModel):
    hypotheses: list[Hypothesis]

    def finalize(self, incident_id: UUID) -> RootCauseReport:
        return RootCauseReport(incident_id=incident_id, **self.model_dump())


class ImpactReportDraft(ContractModel):
    affected_services: list[str]
    candidate_actions: list[CandidateAction]
    summary: str

    def finalize(self, incident_id: UUID) -> ImpactReport:
        return ImpactReport(incident_id=incident_id, **self.model_dump())

from datetime import datetime
from typing import Literal

from pydantic import Field

from app.schemas.base import ContractModel
from app.schemas.enums import ActionType, AgentRole, PermissionLevel, PolicyPreset


class ServiceRule(ContractModel):
    service: str
    tags: list[Literal["stateful", "critical"]] = Field(default_factory=list)
    overrides: dict[ActionType, PermissionLevel] = Field(default_factory=dict)


class Thresholds(ContractModel):
    max_affected_services: int = Field(ge=0)
    max_dependents: int = Field(ge=0)
    approval_risk_score: int = Field(ge=0, le=100)


class ProtectedPair(ContractModel):
    service: str
    action: ActionType


class Protected(ContractModel):
    services: list[str] = Field(default_factory=list)
    actions: list[ActionType] = Field(default_factory=list)
    pairs: list[ProtectedPair] = Field(default_factory=list)


class ApprovalSettings(ContractModel):
    approver_role: str = Field(min_length=1)
    timeout_seconds: int = Field(gt=0)
    on_timeout: Literal["deny"]  # can only be deny (invariant 6)


class PolicyBody(ContractModel):
    name: str = Field(min_length=1)
    preset: PolicyPreset
    # A missing matrix entry means block (invariant 3).
    matrix: dict[AgentRole, dict[ActionType, PermissionLevel]] = Field(default_factory=dict)
    service_rules: list[ServiceRule] = Field(default_factory=list)
    thresholds: Thresholds
    protected: Protected
    approval: ApprovalSettings


class Policy(PolicyBody):
    version: int = Field(ge=1)
    created_by: str
    created_at: datetime
